package "org.example" {
    class Main {
        +main(args: String[]): void
    }
}

package "org.example.factory" {
    class ApplicationFactory {
        +{static} createBank(): Bank
        +{static} createServices(): ServiceBundle
        +{static} createMainMenu(): MainMenuScreen
        +{static} createCommands(bank: Bank, io: ConsoleIO): List<MenuCommand>
    }
    
    class ServiceBundle {
        -customerService: CustomerService
        -accountService: AccountService
        -transactionService: TransactionService
        -reportServiceStrategy: ReportServiceStrategy
        -storageService: StorageService
        +getBank(): Bank
    }
}

package "org.example.service" {
    class Bank {
        +Bank(cs, as, ts, rss, ss)
    }
    
    class CustomerService
    class AccountService
    class TransactionService
    class ReportServiceStrategy
    class StorageService
}

package "org.example.ui" {
    class ConsoleIO
    class MainMenuScreen {
        +MainMenuScreen(commands, io)
    }
    
    interface MenuCommand
}
